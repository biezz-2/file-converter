

# File Converter & Compressor Application with Next.js

I'll help you create a comprehensive file converter and compressor application with all the features you requested. Let's start with a complete plan and implementation.

## 1. Architecture Overview

### Tech Stack
- **Frontend**: Next.js 14 (App Router)
- **UI**: Tailwind CSS, Framer Motion, Three.js for 3D elements
- **Backend**: Next.js API Routes
- **File Processing**: FFmpeg (for videos), Sharp (for images), JSZip (for ZIP files)
- **GPU Acceleration**: WebGPU API for supported browsers, server-side GPU processing
- **State Management**: Zustand
- **File Upload**: React Dropzone
- **UI Components**: Radix UI primitives
- **Database**: PostgreSQL (for user accounts and conversion history)
- **Authentication**: NextAuth.js
- **Deployment**: Vercel or Docker

### System Architecture

### 4.3 Authentication Setup

Create `lib/auth.ts`:

```typescript
import NextAuth from "next-auth"
import { PrismaAdapter } from "@next-auth/prisma-adapter"
import GoogleProvider from "next-auth/providers/google"
import CredentialsProvider from "next-auth/providers/credentials"
import { prisma } from "./db"
import bcrypt from "bcryptjs"

export const authOptions = {
  adapter: PrismaAdapter(prisma),
  providers: [
    GoogleProvider({
      clientId: process.env.GOOGLE_CLIENT_ID!,
      clientSecret: process.env.GOOGLE_CLIENT_SECRET!,
    }),
    CredentialsProvider({
      name: "credentials",
      credentials: {
        email: { label: "Email", type: "email" },
        password: { label: "Password", type: "password" }
      },
      async authorize(credentials) {
        if (!credentials?.email || !credentials?.password) {
          return null
        }

        const user = await prisma.user.findUnique({
          where: { email: credentials.email }
        })

        if (!user || !user.password) {
          return null
        }

        const isPasswordValid = await bcrypt.compare(
          credentials.password,
          user.password
        )

        if (!isPasswordValid) {
          return null
        }

        return {
          id: user.id,
          email: user.email,
          name: user.name,
        }
      }
    })
  ],
  session: {
    strategy: "jwt" as const,
  },
  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        token.id = user.id
      }
      return token
    },
    async session({ session, token }) {
      if (token) {
        session.user.id = token.id as string
      }
      return session
    }
  },
  pages: {
    signIn: "/login",
    signUp: "/register",
  },
}

export default NextAuth(authOptions)
```

### 4.4 3D Background Component

Create `components/3d/Background3D.tsx`:

```typescript
"use client"

import { useEffect, useRef, useState } from "react"
import * as THREE from "three"
import { useTheme } from "next-themes"

export default function Background3D() {
  const mountRef = useRef<HTMLDivElement>(null)
  const [isMounted, setIsMounted] = useState(false)
  const { theme } = useTheme()

  useEffect(() => {
    setIsMounted(true)
    return () => setIsMounted(false)
  }, [])

  useEffect(() => {
    if (!isMounted || !mountRef.current) return

    const width = window.innerWidth
    const height = window.innerHeight

    // Scene setup
    const scene = new THREE.Scene()
    const camera = new THREE.PerspectiveCamera(75, width / height, 0.1, 1000)
    const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true })
    
    renderer.setSize(width, height)
    renderer.setPixelRatio(window.devicePixelRatio)
    renderer.setClearColor(0x000000, 0)
    mountRef.current.appendChild(renderer.domElement)

    // Lighting
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.5)
    scene.add(ambientLight)
    
    const directionalLight = new THREE.DirectionalLight(0xffffff, 0.5)
    directionalLight.position.set(0, 1, 1)
    scene.add(directionalLight)

    // Create floating particles
    const particlesGeometry = new THREE.BufferGeometry()
    const particlesCount = 500
    const posArray = new Float32Array(particlesCount * 3)
    
    for(let i = 0; i < particlesCount * 3; i++) {
      posArray[i] = (Math.random() - 0.5) * 10
    }
    
    particlesGeometry.setAttribute('position', new THREE.BufferAttribute(posArray, 3))
    
    const particlesMaterial = new THREE.PointsMaterial({
      size: 0.05,
      color: theme === 'dark' ? 0x3b82f6 : 0x8b5cf6,
      transparent: true,
      opacity: 0.8,
      blending: THREE.AdditiveBlending
    })
    
    const particlesMesh = new THREE.Points(particlesGeometry, particlesMaterial)
    scene.add(particlesMesh)

    // Create floating geometric shapes
    const geometries = [
      new THREE.BoxGeometry(0.5, 0.5, 0.5),
      new THREE.ConeGeometry(0.3, 0.6, 8),
      new THREE.TorusGeometry(0.3, 0.1, 8, 20),
      new THREE.OctahedronGeometry(0.4)
    ]
    
    const materials = [
      new THREE.MeshStandardMaterial({ 
        color: 0x3b82f6, 
        metalness: 0.3, 
        roughness: 0.4,
        transparent: true,
        opacity: 0.7
      }),
      new THREE.MeshStandardMaterial({ 
        color: 0x8b5cf6, 
        metalness: 0.3, 
        roughness: 0.4,
        transparent: true,
        opacity: 0.7
      }),
      new THREE.MeshStandardMaterial({ 
        color: 0x10b981, 
        metalness: 0.3, 
        roughness: 0.4,
        transparent: true,
        opacity: 0.7
      })
    ]
    
    const shapes: THREE.Mesh[] = []
    
    for (let i = 0; i < 8; i++) {
      const geometry = geometries[Math.floor(Math.random() * geometries.length)]
      const material = materials[Math.floor(Math.random() * materials.length)]
      const shape = new THREE.Mesh(geometry, material)
      
      shape.position.x = (Math.random() - 0.5) * 8
      shape.position.y = (Math.random() - 0.5) * 8
      shape.position.z = (Math.random() - 0.5) * 8
      
      shape.rotation.x = Math.random() * Math.PI
      shape.rotation.y = Math.random() * Math.PI
      
      scene.add(shape)
      shapes.push(shape)
    }

    camera.position.z = 5

    // Mouse movement effect
    let mouseX = 0
    let mouseY = 0
    
    const handleMouseMove = (event: MouseEvent) => {
      mouseX = (event.clientX / width) * 2 - 1
      mouseY = -(event.clientY / height) * 2 + 1
    }
    
    window.addEventListener('mousemove', handleMouseMove)

    // Animation
    const clock = new THREE.Clock()
    
    const animate = () => {
      requestAnimationFrame(animate)
      
      const elapsedTime = clock.getElapsedTime()
      
      // Rotate particles
      particlesMesh.rotation.y = elapsedTime * 0.05
      
      // Animate shapes
      shapes.forEach((shape, index) => {
        shape.rotation.x += 0.005
        shape.rotation.y += 0.005
        
        // Float up and down
        shape.position.y += Math.sin(elapsedTime + index) * 0.003
      })
      
      // Camera follows mouse slightly
      camera.position.x += (mouseX * 0.5 - camera.position.x) * 0.05
      camera.position.y += (mouseY * 0.5 - camera.position.y) * 0.05
      camera.lookAt(scene.position)
      
      renderer.render(scene, camera)
    }
    
    animate()

    // Handle resize
    const handleResize = () => {
      const width = window.innerWidth
      const height = window.innerHeight
      
      camera.aspect = width / height
      camera.updateProjectionMatrix()
      
      renderer.setSize(width, height)
    }
    
    window.addEventListener('resize', handleResize)

    // Cleanup
    return () => {
      window.removeEventListener('mousemove', handleMouseMove)
      window.removeEventListener('resize', handleResize)
      
      if (mountRef.current && renderer.domElement) {
        mountRef.current.removeChild(renderer.domElement)
      }
      
      renderer.dispose()
    }
  }, [isMounted, theme])

  return (
    <div 
      ref={mountRef} 
      className="fixed inset-0 -z-10 overflow-hidden" 
      style={{ pointerEvents: 'none' }}
    />
  )
}
```

### 4.5 File Processing Utilities

Create `lib/file-processing.ts`:

```typescript
import sharp from 'sharp'
import ffmpeg from 'fluent-ffmpeg'
import ffmpegStatic from 'ffmpeg-static'
import JSZip from 'jszip'
import path from 'path'
import fs from 'fs/promises'

// Set FFmpeg path
ffmpeg.setFfmpegPath(ffmpegStatic as string)

export interface ConversionOptions {
  format?: string
  quality?: number
  width?: number
  height?: number
  compression?: number
  fps?: number
  bitrate?: string
}

export interface ConversionResult {
  success: boolean
  outputPath?: string
  error?: string
  originalSize?: number
  newSize?: number
}

// Image processing
export async function processImage(
  inputPath: string,
  outputPath: string,
  options: ConversionOptions
): Promise<ConversionResult> {
  try {
    const inputBuffer = await fs.readFile(inputPath)
    const originalSize = inputBuffer.length

    let sharpInstance = sharp(inputBuffer)

    // Resize if dimensions are provided
    if (options.width || options.height) {
      sharpInstance = sharpInstance.resize(
        options.width,
        options.height,
        { fit: 'inside', withoutEnlargement: true }
      )
    }

    // Set format
    if (options.format) {
      sharpInstance = sharpInstance.toFormat(options.format as any, {
        quality: options.quality || 80,
        compressionLevel: options.compression || 6
      })
    } else {
      // Default to JPEG with quality
      sharpInstance = sharpInstance.jpeg({ quality: options.quality || 80 })
    }

    const outputBuffer = await sharpInstance.toBuffer()
    await fs.writeFile(outputPath, outputBuffer)

    return {
      success: true,
      outputPath,
      originalSize,
      newSize: outputBuffer.length
    }
  } catch (error) {
    console.error('Image processing error:', error)
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error'
    }
  }
}

// Video processing
export async function processVideo(
  inputPath: string,
  outputPath: string,
  options: ConversionOptions
): Promise<ConversionResult> {
  return new Promise((resolve) => {
    try {
      // Get original file size
      fs.stat(inputPath)
        .then(stats => {
          const originalSize = stats.size

          let command = ffmpeg(inputPath)

          // Set output format
          if (options.format) {
            command = command.format(options.format)
          }

          // Set video codec
          command = command.videoCodec('libx264')

          // Set dimensions
          if (options.width || options.height) {
            command = command.size(`${options.width || '?'}x${options.height || '?'}`)
          }

          // Set quality
          if (options.quality) {
            command = command.videoBitrate(`${options.quality}k`)
          }

          // Set framerate
          if (options.fps) {
            command = command.fps(options.fps)
          }

          // Set bitrate
          if (options.bitrate) {
            command = command.bitrate(options.bitrate)
          }

          command
            .output(outputPath)
            .on('end', async () => {
              try {
                const outputStats = await fs.stat(outputPath)
                resolve({
                  success: true,
                  outputPath,
                  originalSize,
                  newSize: outputStats.size
                })
              } catch (error) {
                resolve({
                  success: false,
                  error: 'Failed to get output file stats'
                })
              }
            })
            .on('error', (err) => {
              console.error('Video processing error:', err)
              resolve({
                success: false,
                error: err.message
              })
            })
            .run()
        })
        .catch(error => {
          console.error('Failed to get input file stats:', error)
          resolve({
            success: false,
            error: 'Failed to get input file stats'
          })
        })
    } catch (error) {
      console.error('Video processing setup error:', error)
      resolve({
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error'
      })
    }
  })
}

// Archive processing
export async function processArchive(
  inputPath: string,
  outputPath: string,
  options: ConversionOptions
): Promise<ConversionResult> {
  try {
    const inputBuffer = await fs.readFile(inputPath)
    const originalSize = inputBuffer.length

    const zip = new JSZip()
    await zip.loadAsync(inputBuffer)

    // If compression level is specified, we need to recreate the zip with new compression
    if (options.compression !== undefined) {
      const newZip = new JSZip()
      
      // Add all files with new compression level
      for (const [filename, file] of Object.entries(zip.files)) {
        if (!file.dir) {
          const content = await file.async('uint8array')
          newZip.file(filename, content, { compression: 'DEFLATE', compressionOptions: { level: options.compression } })
        }
      }
      
      const outputBuffer = await newZip.generateAsync({ type: 'uint8array' })
      await fs.writeFile(outputPath, Buffer.from(outputBuffer))
      
      return {
        success: true,
        outputPath,
        originalSize,
        newSize: outputBuffer.length
      }
    } else {
      // Just copy the file if no compression options
      await fs.writeFile(outputPath, inputBuffer)
      
      return {
        success: true,
        outputPath,
        originalSize,
        newSize: originalSize
      }
    }
  } catch (error) {
    console.error('Archive processing error:', error)
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error'
    }
  }
}

// Get file type
export function getFileType(filename: string): 'image' | 'video' | 'archive' | 'document' | 'other' {
  const ext = path.extname(filename).toLowerCase()
  
  const imageExts = ['.jpg', '.jpeg', '.png', '.gif', '.bmp', '.webp', '.tiff', '.svg']
  const videoExts = ['.mp4', '.avi', '.mov', '.wmv', '.flv', '.webm', '.mkv', '.m4v']
  const archiveExts = ['.zip', '.rar', '.7z', '.tar', '.gz']
  const documentExts = ['.pdf', '.doc', '.docx', '.xls', '.xlsx', '.ppt', '.pptx', '.txt']
  
  if (imageExts.includes(ext)) return 'image'
  if (videoExts.includes(ext)) return 'video'
  if (archiveExts.includes(ext)) return 'archive'
  if (documentExts.includes(ext)) return 'document'
  
  return 'other'
}

// Get supported output formats for each file type
export function getSupportedFormats(fileType: string): string[] {
  switch (fileType) {
    case 'image':
      return ['jpg', 'jpeg', 'png', 'webp', 'gif', 'bmp', 'tiff']
    case 'video':
      return ['mp4', 'avi', 'mov', 'webm', 'mkv', 'flv']
    case 'archive':
      return ['zip']
    case 'document':
      return ['pdf']
    default:
      return []
  }
}
```

### 4.6 GPU Acceleration Utilities

```typescript
// Client-side GPU acceleration using WebGPU
export async function checkWebGPUSupport(): Promise<boolean> {
  if (!navigator.gpu) {
    return false
  }
  
  try {
    const adapter = await navigator.gpu.requestAdapter()
    return !!adapter
  } catch (e) {
    return false
  }
}

// Server-side GPU acceleration detection
export function checkServerGPU(): boolean {
  try {
    // This would need to be implemented based on your server environment
    // For example, checking for NVIDIA GPUs via nvidia-smi or other methods
    return process.env.GPU_AVAILABLE === 'true'
  } catch (e) {
    return false
  }
}

// Client-side image processing with WebGPU (simplified example)
export async function processImageWithGPU(
  imageData: ImageData,
  width: number,
  height: number,
  options: { brightness?: number; contrast?: number } = {}
): Promise<ImageData> {
  if (!navigator.gpu) {
    throw new Error('WebGPU not supported')
  }
  
  const adapter = await navigator.gpu.requestAdapter()
  if (!adapter) {
    throw new Error('No appropriate GPUAdapter found')
  }
  
  const device = await adapter.requestDevice()
  
  // Create shader module
  const shaderModule = device.createShaderModule({
    code: `
      @group(0) @binding(0) var<uniform> params: vec2f;
      @group(0) @binding(1) var<storage, read> inputImage: array<vec4f>;
      @group(0) @binding(2) var<storage, read_write> outputImage: array<vec4f>;
      
      @compute @workgroup_size(64)
      fn main(@builtin(global_invocation_id) global_id: vec3<u32>) {
        let index = global_id.x;
        if (index >= arrayLength(&outputImage)) {
          return;
        }
        
        var color = inputImage[index];
        color.rgb = color.rgb * params.x + params.y;
        outputImage[index] = color;
      }
    `
  });
  
  // Create pipeline
  const pipeline = device.createComputePipeline({
    layout: 'auto',
    compute: {
      module: shaderModule,
      entryPoint: 'main',
    },
  });
  
  // Create buffers
  const imageSize = width * height;
  const inputBuffer = device.createBuffer({
    size: imageSize * 4 * 4, // 4 components (RGBA) * 4 bytes each
    usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST,
    mappedAtCreation: true,
  });
  
  // Copy image data to buffer
  const inputArray = new Uint8Array(imageData.data);
  new Uint8Array(inputBuffer.getMappedRange()).set(inputArray);
  inputBuffer.unmap();
  
  const outputBuffer = device.createBuffer({
    size: imageSize * 4 * 4,
    usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_SRC,
  });
  
  // Create uniform buffer for parameters
  const uniformBuffer = device.createBuffer({
    size: 2 * 4, // 2 floats (brightness, contrast) * 4 bytes each
    usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST,
  });
  
  // Set parameters
  const params = new Float32Array([
    options.contrast || 1.0,
    options.brightness || 0.0,
  ]);
  device.queue.writeBuffer(uniformBuffer, 0, params);
  
  // Create bind group
  const bindGroup = device.createBindGroup({
    layout: pipeline.getBindGroupLayout(0),
    entries: [
      { binding: 0, resource: { buffer: uniformBuffer } },
      { binding: 1, resource: { buffer: inputBuffer } },
      { binding: 2, resource: { buffer: outputBuffer } },
    ],
  });
  
  // Execute shader
  const commandEncoder = device.createCommandEncoder();
  const passEncoder = commandEncoder.beginComputePass();
  passEncoder.setPipeline(pipeline);
  passEncoder.setBindGroup(0, bindGroup);
  passEncoder.dispatchWorkgroups(Math.ceil(imageSize / 64));
  passEncoder.end();
  
  // Read back results
  const readbackBuffer = device.createBuffer({
    size: imageSize * 4 * 4,
    usage: GPUBufferUsage.COPY_DST | GPUBufferUsage.MAP_READ,
  });
  
  commandEncoder.copyBufferToBuffer(outputBuffer, 0, readbackBuffer, 0, imageSize * 4 * 4);
  
  device.queue.submit([commandEncoder.finish()]);
  
  // Wait for completion and get results
  await readbackBuffer.mapAsync(GPUMapMode.READ);
  const resultArray = new Uint8Array(readbackBuffer.getMappedRange());
  
  // Create new ImageData
  const outputImageData = new ImageData(
    new Uint8ClampedArray(resultArray),
    width,
    height
  );
  
  readbackBuffer.unmap();
  
  return outputImageData;
}
```

```typescript
"use client"

import { useCallback, useState } from 'react'
import { useDropzone } from 'react-dropzone'
import { motion, AnimatePresence } from 'framer-motion'
import { File, X, Upload, Image, Video, Archive, FileText } from 'lucide-react'
import { getFileType } from '@/lib/file-processing'

interface FileUploaderProps {
  onFilesChange: (files: File[]) => void
  maxFiles?: number
  maxSize?: number // in bytes
}

export default function FileUploader({ onFilesChange, maxFiles = 10, maxSize = 100 * 1024 * 1024 }: FileUploaderProps) {
  const [files, setFiles] = useState<File[]>([])
  const [error, setError] = useState<string | null>(null)

  const onDrop = useCallback((acceptedFiles: File[], fileRejections: any[]) => {
    setError(null)
    
    if (fileRejections.length > 0) {
      setError('Some files were rejected. Please check file size and type.')
      return
    }
    
    const newFiles = [...files, ...acceptedFiles].slice(0, maxFiles)
    setFiles(newFiles)
    onFilesChange(newFiles)
  }, [files, maxFiles, onFilesChange])

  const removeFile = (index: number) => {
    const newFiles = files.filter((_, i) => i !== index)
    setFiles(newFiles)
    onFilesChange(newFiles)
  }

  const { getRootProps, getInputProps, isDragActive } = useDropzone({
    onDrop,
    accept: {
      'image/*': ['.jpeg', '.jpg', '.png', '.gif', '.webp', '.bmp', '.tiff'],
      'video/*': ['.mp4', '.avi', '.mov', '.wmv', '.flv', '.webm', '.mkv', '.m4v'],
      'application/zip': ['.zip'],
      'application/pdf': ['.pdf'],
      'application/msword': ['.doc'],
      'application/vnd.openxmlformats-officedocument.wordprocessingml.document': ['.docx'],
      'application/vnd.ms-excel': ['.xls'],
      'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet': ['.xlsx'],
      'application/vnd.ms-powerpoint': ['.ppt'],
      'application/vnd.openxmlformats-officedocument.presentationml.presentation': ['.pptx'],
      'text/plain': ['.txt']
    },
    maxSize,
    multiple: true
  })

  const getFileIcon = (file: File) => {
    const type = getFileType(file.name)
    
    switch (type) {
      case 'image':
        return <Image className="w-5 h-5" />
      case 'video':
        return <Video className="w-5 h-5" />
      case 'archive':
        return <Archive className="w-5 h-5" />
      case 'document':
        return <FileText className="w-5 h-5" />
      default:
        return <File className="w-5 h-5" />
    }
  }

  const formatFileSize = (bytes: number) => {
    if (bytes === 0) return '0 Bytes'
    const k = 1024
    const sizes = ['Bytes', 'KB', 'MB', 'GB']
    const i = Math.floor(Math.log(bytes) / Math.log(k))
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i]
  }

  return (
    <div className="w-full">
      <div
        {...getRootProps()}
        className={`border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors ${
          isDragActive
            ? 'border-blue-500 bg-blue-50 dark:bg-blue-950/20'
            : 'border-gray-300 dark:border-gray-700 hover:border-gray-400 dark:hover:border-gray-600'
        }`}
      >
        <input {...getInputProps()} />
        <motion.div
          initial={{ opacity: 0, y: 10 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.3 }}
        >
          <Upload className="mx-auto h-12 w-12 text-gray-400 mb-4" />
          <p className="text-lg font-medium text-gray-900 dark:text-gray-100 mb-2">
            {isDragActive ? 'Drop files here' : 'Drag & drop files here, or click to select files'}
          </p>
          <p className="text-sm text-gray-500 dark:text-gray-400">
            Supported formats: Images, Videos, Archives, Documents
          </p>
          <p className="text-xs text-gray-400 dark:text-gray-500 mt-2">
            Max file size: {formatFileSize(maxSize)} | Max files: {maxFiles}
          </p>
        </motion.div>
      </div>

      {error && (
        <motion.div
          initial={{ opacity: 0, y: -10 }}
          animate={{ opacity: 1, y: 0 }}
          className="mt-4 p-3 bg-red-50 dark:bg-red-950/20 border border-red-200 dark:border-red-800 rounded-md"
        >
          <p className="text-sm text-red-600 dark:text-red-400">{error}</p>
        </motion.div>
      )}

      <AnimatePresence>
        {files.length > 0 && (
          <motion.div
            initial={{ opacity: 0, height: 0 }}
            animate={{ opacity: 1, height: 'auto' }}
            exit={{ opacity: 0, height: 0 }}
            className="mt-6"
          >
            <h3 className="text-lg font-medium text-gray-900 dark:text-gray-100 mb-3">
              Selected Files ({files.length})
            </h3>
            <div className="space-y-2">
              {files.map((file, index) => (
                <motion.div
                  key={`${file.name}-${index}`}
                  initial={{ opacity: 0, x: -20 }}
                  animate={{ opacity: 1, x: 0 }}
                  exit={{ opacity: 0, x: 20 }}
                  transition={{ duration: 0.2 }}
                  className="flex items-center justify-between p-3 bg-white dark:bg-gray-800 rounded-md shadow-sm border border-gray-200 dark:border-gray-700"
                >
                  <div className="flex items-center space-x-3">
                    <div className="text-blue-500 dark:text-blue-400">
                      {getFileIcon(file)}

                    <div>
```

```typescript
"use client"

import { motion } from 'framer-motion'
import { CheckCircle, XCircle, AlertCircle } from 'lucide-react'

interface ConversionProgress {
  id: string
  filename: string
  status: 'pending' | 'processing' | 'completed' | 'failed'
  progress: number
  error?: string
  originalSize?: number
  newSize?: number
}

interface ProgressBarProps {
  conversions: ConversionProgress[]
}

export default function ProgressBar({ conversions }: ProgressBarProps) {
  const formatFileSize = (bytes?: number) => {
    if (!bytes) return '0 Bytes'
    const k = 1024
    const sizes = ['Bytes', 'KB', 'MB', 'GB']
    const i = Math.floor(Math.log(bytes) / Math.log(k))
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i]
  }

  const getStatusIcon = (status: string) => {
    switch (status) {
      case 'completed':
        return <CheckCircle className="w-5 h-5 text-green-500" />
      case 'failed':
        return <XCircle className="w-5 h-5 text-red-500" />
      case 'processing':
        return (
          <motion.div
            className="w-5 h-5 rounded-full border-2 border-blue-500 border-t-transparent"
            animate={{ rotate: 360 }}
            transition={{ duration: 1, repeat: Infinity, ease: "linear" }}
          />
        )
      default:
        return <AlertCircle className="w-5 h-5 text-gray-400" />
    }
  }

  if (conversions.length === 0) {
    return null
  }

  return (
    <div className="space-y-4">
      {conversions.map((conversion) => (
        <motion.div
          key={conversion.id}
          initial={{ opacity: 0, y: 10 }}
          animate={{ opacity: 1, y: 0 }}
          className="bg-white dark:bg-gray-800 rounded-lg shadow-md p-4"
        >
          <div className="flex items-center justify-between mb-2">
            <div className="flex items-center">
              {getStatusIcon(conversion.status)}
              <span className="ml-2 text-sm font-medium text-gray-900 dark:text-gray-100 truncate max-w-xs">
                {conversion.filename}
              </span>
            </div>
            <span className="text-xs text-gray-500 dark:text-gray-400">
              {conversion.progress}%
            </span>
          </div>
          
          <div className="w-full bg-gray-200 dark:bg-gray-700 rounded-full h-2 mb-2">
            <motion.div
              className="bg-blue-600 h-2 rounded-full"
              initial={{ width: 0 }}
              animate={{ width: `${conversion.progress}%` }}
              transition={{ duration: 0.3 }}
            />
          </div>
          
          {conversion.error && (
            <motion.div
              initial={{ opacity: 0, height: 0 }}
              animate={{ opacity: 1, height: 'auto' }}
              className="mt-2 p-2 bg-red-50 dark:bg-red-950/20 rounded"
            >
              <p className="text-xs text-red-600 dark:text-red-400">{conversion.error}</p>
```

### 4.11 Main Converter Page

Create `app/converter/page.tsx`:

```typescript
"use client"

import { useState, useEffect, useRef } from 'react'
import { motion } from 'framer-motion'
import { useSession } from 'next-auth/react'
import { useRouter } from 'next/navigation'
import { toast } from 'react-hot-toast'

import FileUploader from './components/FileUploader'
import ConversionOptions from './components/ConversionOptions'
import ProgressBar from './components/ProgressBar'
import ResultDisplay from './components/ResultDisplay'

import Background3D from '@/components/3d/Background3D'
import { processImage, processVideo, processArchive, getFileType } from '@/lib/file-processing'

interface ConversionProgress {
  id: string
  filename: string
  status: 'pending' | 'processing' | 'completed' | 'failed'
  progress: number
  error?: string
  originalSize?: number
  newSize?: number
}

interface ConversionResult {
  id: string
  filename: string
  originalFilename: string
  url: string
  size: number
  originalSize: number
  type: string
}

export default function ConverterPage() {
  const { data: session, status } = useSession()
  const router = useRouter()
  
  const [files, setFiles] = useState<File[]>([])
  const [conversions, setConversions] = useState<ConversionProgress[]>([])
  const [results, setResults] = useState<ConversionResult[]>([])
  const [isConverting, setIsConverting] = useState(false)
  const fileInputRef = useRef<HTMLInputElement>(null)
  
  useEffect(() => {
    if (status === 'unauthenticated') {
      router.push('/login')
    }
  }, [status, router])

  const handleFilesChange = (newFiles: File[]) => {
    setFiles(newFiles)
    setResults([])
  }

  const handleConvert = async (options: any) => {
    if (files.length === 0) return
    
    setIsConverting(true)
    setResults([])
    
    // Initialize conversion progress for each file
    const newConversions: ConversionProgress[] = files.map(file => ({
      id: Math.random().toString(36).substring(2, 9),
      filename: file.name,
      status: 'pending',
      progress: 0,
      originalSize: file.size
    }))
    
    setConversions(newConversions)
    
    // Process files one by one (in a real app, you might want to process them in parallel)
    for (let i = 0; i < files.length; i++) {
      const file = files[i]
      const conversionId = newConversions[i].id
      
      try {
        // Update status to processing
        setConversions(prev => 
          prev.map(c => 
            c.id === conversionId 
              ? { ...c, status: 'processing', progress: 10 }
              : c
          )
        )
        
        // Create form data for API call
        const formData = new FormData()
        formData.append('file', file)
        formData.append('options', JSON.stringify(options))
        
        // Determine which API endpoint to use based on file type
        const fileType = getFileType(file.name)
        let endpoint = ''
        
        switch (fileType) {
          case 'image':
            endpoint = '/api/convert/image'
            break
          case 'video':
            endpoint = '/api/convert/video'
            break
          case 'archive':
            endpoint = '/api/compress/archive'
            break
          default:
            endpoint = '/api/convert/document'
        }
        
        // Update progress
        setConversions(prev => 
          prev.map(c => 
            c.id === conversionId 
              ? { ...c, progress: 30 }
              : c
          )
        )
        
        // Call API
        const response = await fetch(endpoint, {
          method: 'POST',
          body: formData
        })
        
        if (!response.ok) {
          throw new Error('Conversion failed')
        }
        
        // Update progress
        setConversions(prev => 
          prev.map(c => 
            c.id === conversionId 
              ? { ...c, progress: 70 }
              : c
          )
        )
        
        const result = await response.json()
        
        // Update progress to completed
        setConversions(prev => 
          prev.map(c => 
            c.id === conversionId 
              ? { ...c, status: 'completed', progress: 100, newSize: result.size }
              : c
          )
        )
        
        // Add to results
        setResults(prev => [...prev, {
          id: conversionId,
          filename: result.filename,
          originalFilename: file.name,
          url: result.url,
          size: result.size,
          originalSize: file.size,
          type: result.type
        }])
        
        toast.success(`${file.name} converted successfully!`)
      } catch (error) {
        console.error('Conversion error:', error)
        
        // Update status to failed
        setConversions(prev => 
          prev.map(c => 
            c.id === conversionId 
              ? { 
                  ...c, 
                  status: 'failed', 
                  error: error instanceof Error ? error.message : 'Unknown error'
                }
              : c
          )
        )
        
        toast.error(`Failed to convert ${file.name}`)
      }
    }
    
    setIsConverting(false)
  }

  const handleDownload = (id: string) => {
    const result = results.find(r => r.id === id)
    if (!result) return
    
    // Create a temporary link and trigger download
    const link = document.createElement('a')
    link.href = result.url
    link.download = result.filename
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
  }

  const handlePreview = (id: string) => {
    const result = results.find(r => r.id === id)
    if (!result || !result.type.startsWith('image/')) return
    
    // Open preview in a new tab
    window.open(result.url, '_blank')
  }

  const handleClear = () => {
    setResults([])
    setConversions([])
    setFiles([])
  }

  if (status === 'loading') {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="animate-spin rounded-full h-12 w-12 border-t-2 border-b-2 border-blue-500"></div>
      </div>
    )
  }

  return (
    <div className="min-h-screen bg-gray-50 dark:bg-gray-900 relative overflow-hidden">
      <Background3D />
      
      <div className="container mx-auto px-4 py-8 relative z-10">
        <motion.div
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.5 }}
          className="text-center mb-10"
        >
          <h1 className="text-4xl font-bold text-gray-900 dark:text-white mb-4">
            File Converter & Compressor
          </h1>
          <p className="text-lg text-gray-600 dark:text-gray-300 max-w-2xl mx-auto">
            Convert and compress your files with ease. Supports images, videos, archives, and documents.
            Powered by GPU acceleration for faster processing.
          </p>
        </motion.div>
        
        <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
          <div className="lg:col-span-2 space-y-8">
            <FileUploader onFilesChange={handleFilesChange} />
            
            <ProgressBar conversions={conversions} />
            
            <ResultDisplay
              results={results}
              onClear={handleClear}
              onDownload={handleDownload}
              onPreview={handlePreview}
            />
```

### 4.12 API Routes for File Processing

Create `app/api/convert/image/route.ts`:

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { writeFile, mkdir } from 'fs/promises'
import { join } from 'path'
import { processImage } from '@/lib/file-processing'
import { v4 as uuidv4 } from 'uuid'

export async function POST(request: NextRequest) {
  try {
    const data = await request.formData()
    const file: File | null = data.get('file') as unknown as File
    const options = JSON.parse(data.get('options') as string)
    
    if (!file) {
      return NextResponse.json({ error: 'No file provided' }, { status: 400 })
    }
    
    const bytes = await file.arrayBuffer()
    const buffer = Buffer.from(bytes)
    
    // Create a unique filename
    const fileId = uuidv4()
    const inputPath = join(process.cwd(), 'temp', `${fileId}-input${file.name.substring(file.name.lastIndexOf('.'))}`)
    const outputPath = join(process.cwd(), 'temp', `${fileId}-output.${options.outputFormat || 'jpg'}`)
    
    // Ensure temp directory exists
    await mkdir(join(process.cwd(), 'temp'), { recursive: true })
    
    // Write the file to disk
    await writeFile(inputPath, buffer)
    
    // Process the image
    const result = await processImage(inputPath, outputPath, options)
    
    if (!result.success) {
      return NextResponse.json({ error: result.error }, { status: 500 })
    }
    
    // Read the processed file
    const outputBuffer = await readFile(outputPath)
    
    // Clean up temp files
    await unlink(inputPath)
    await unlink(outputPath)
    
    // Return the processed file
    return new NextResponse(outputBuffer, {
      headers: {
        'Content-Type': `image/${options.outputFormat || 'jpg'}`,
        'Content-Disposition': `attachment; filename="${file.name.substring(0, file.name.lastIndexOf('.'))}.${options.outputFormat || 'jpg'}"`,
      },
    })
  } catch (error) {
    console.error('Image conversion error:', error)
    return NextResponse.json(
      { error: 'Failed to convert image' },
      { status: 500 }
    )
  }
}
```

### 5.1 Docker Setup

Create `Dockerfile`:

```dockerfile
# Install dependencies only when needed
FROM node:18-alpine AS deps
# Check https://github.com/nodejs/docker-node/tree/b4117f9333da4138b03a546ec926ef50a31506c3#nodealpine to understand why libc6-compat might be needed.
RUN apk add --no-cache libc6-compat
WORKDIR /app

# Install dependencies based on the preferred package manager
COPY package.json yarn.lock* package-lock.json* pnpm-lock.yaml* ./
RUN \
  if [ -f yarn.lock ]; then yarn --frozen-lockfile; \
  elif [ -f package-lock.json ]; then npm ci; \
  elif [ -f pnpm-lock.yaml ]; then yarn global add pnpm && pnpm i --frozen-lockfile; \
  else echo "Lockfile not found." && exit 1; \
  fi

# Rebuild the source code only when needed
FROM node:18-alpine AS builder
WORKDIR /app
COPY --from=deps /app/node_modules ./node_modules
COPY . .

# Install FFmpeg for video processing
RUN apk add --no-cache ffmpeg

# Next.js collects completely anonymous telemetry data about general usage.
# Learn more here: https://nextjs.org/telemetry
# Uncomment the following line in case you want to disable telemetry during the build.
# ENV NEXT_TELEMETRY_DISABLED 1

RUN \
  if [ -f yarn.lock ]; then yarn run build; \
  elif [ -f package-lock.json ]; then npm run build; \
  elif [ -f pnpm-lock.yaml ]; then yarn global add pnpm && pnpm run build; \
  else echo "Lockfile not found." && exit 1; \
  fi

# Production image, copy all the files and run next
FROM node:18-alpine AS runner
WORKDIR /app

ENV NODE_ENV production
# Uncomment the following line in case you want to disable telemetry during runtime.
# ENV NEXT_TELEMETRY_DISABLED 1

# Install FFmpeg for video processing
RUN apk add --no-cache ffmpeg

RUN addgroup --system --gid 1001 nodejs
RUN adduser --system --uid 1001 nextjs

COPY --from=builder /app/public ./public

# Set the correct permission for prerender cache
RUN mkdir .next
RUN chown nextjs:nodejs .next

# Automatically leverage output traces to reduce image size
# https://nextjs.org/docs/advanced-features/output-file-tracing
COPY --from=builder --chown=nextjs:nodejs /app/.next/standalone ./
COPY --from=builder --chown=nextjs:nodejs /app/.next/static ./.next/static

USER nextjs

EXPOSE 3000

ENV PORT 3000
# set hostname to localhost
ENV HOSTNAME "0.0.0.0"

# server.js is created by next build from the standalone output
# https://nextjs.org/docs/pages/api-reference/next-config-js/output
CMD ["node", "server.js"]
```

   - Push your code to a GitHub repository
   - Connect your Vercel account to the repository
   - Configure environment variables in Vercel dashboard
   - Deploy

   - Build the application: `npm run build`
   - Start the production server: `npm start`
