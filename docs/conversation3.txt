## 3. YouTube Downloader Implementation

Let's create a robust YouTube downloader service that supports 8K resolution and 320kbps audio:

```typescript
// lib/youtube-downloader.ts
import { exec } from 'child_process';
import { promisify } from 'util';
import path from 'path';
import fs from 'fs/promises';
import { v4 as uuidv4 } from 'uuid';
import { defaultEnterpriseConfig } from './enterprise-architecture';

const execAsync = promisify(exec);

export interface YouTubeVideoInfo {
  id: string;
  title: string;
  duration: number;
  thumbnail: string;
  formats: {
    video: Array<{
      resolution: string;
      fps: number;
      codec: string;
      filesize: number;
    }>;
    audio: Array<{
      bitrate: number;
      codec: string;
      filesize: number;
    }>;
  };
}

export interface DownloadOptions {
  format: 'video' | 'audio';
  resolution?: string;
  audioQuality?: number; // kbps
  outputFormat: string;
}

export class YouTubeDownloader {
  private config = defaultEnterpriseConfig.youtubeConfig;
  private downloadPath: string;

  constructor() {
    this.downloadPath = this.config.downloadPath;
    this.ensureDownloadDirectory();
  }

  private async ensureDownloadDirectory() {
    try {
      await fs.access(this.downloadPath);
    } catch (error) {
      await fs.mkdir(this.downloadPath, { recursive: true });
    }
  }

  async getVideoInfo(url: string): Promise<YouTubeVideoInfo> {
    const tempFile = path.join(this.downloadPath, `${uuidv4()}.json`);
    
    try {
      // Using yt-dlp to get video information
      const command = `yt-dlp --dump-json "${url}" > "${tempFile}"`;
      await execAsync(command);
      
      const data = await fs.readFile(tempFile, 'utf8');
      const info = JSON.parse(data);
      
      // Extract video formats
      const videoFormats = info.formats
        .filter((f: any) => f.vcodec !== 'none' && f.height)
        .map((f: any) => ({
          resolution: `${f.height}p`,
          fps: f.fps,
          codec: f.vcodec,
          filesize: f.filesize || 0,
        }))
        .filter((f: any, i: number, arr: any[]) => 
          arr.findIndex((item: any) => item.resolution === f.resolution) === i
        )
        .sort((a: any, b: any) => {
          const aHeight = parseInt(a.resolution.replace('p', ''));
          const bHeight = parseInt(b.resolution.replace('p', ''));
          return bHeight - aHeight;
        });

      // Extract audio formats
      const audioFormats = info.formats
        .filter((f: any) => f.acodec !== 'none' && f.vcodec === 'none')
        .map((f: any) => ({
          bitrate: f.abr || 0,
          codec: f.acodec,
          filesize: f.filesize || 0,
        }))
        .sort((a: any, b: any) => b.bitrate - a.bitrate);

      return {
        id: info.id,
        title: info.title,
        duration: info.duration,
        thumbnail: info.thumbnail,
        formats: {
          video: videoFormats,
          audio: audioFormats,
        },
      };
    } catch (error) {
      console.error('Error getting video info:', error);
      throw new Error('Failed to retrieve video information');
    } finally {
      // Clean up temp file
      try {
        await fs.unlink(tempFile);
      } catch (error) {
        // Ignore cleanup errors
      }
    }
  }

  async downloadVideo(
    url: string, 
    options: DownloadOptions, 
    onProgress?: (progress: number) => void
  ): Promise<{ filePath: string; fileSize: number }> {
    const jobId = uuidv4();
    const outputFile = path.join(this.downloadPath, `${jobId}.${options.outputFormat}`);
    
    try {
      // Build yt-dlp command
      let command = `yt-dlp --no-warnings --progress --newline`;
      
      // Add cookies if available
      if (this.config.cookiesPath) {
        command += ` --cookies "${this.config.cookiesPath}"`;
      }
      
      // Add format selector
      if (options.format === 'video') {
        const resolution = options.resolution || '1080p';
        command += ` -f "bestvideo[height<=${resolution.replace('p', '')}]+bestaudio/best[height<=${resolution.replace('p', '')}]"`;
      } else {
        const bitrate = options.audioQuality || this.config.maxAudioBitrate;
        command += ` -f "bestaudio[abr<=${bitrate}]/bestaudio"`;
      }
      
      // Add output format
      if (options.format === 'audio') {
        command += ` -x --audio-format ${options.outputFormat}`;
        if (options.audioQuality) {
          command += ` --audio-quality ${options.audioQuality}`;
        }
      } else {
        command += ` --merge-output-format ${options.outputFormat}`;
      }
      
      // Add output file
      command += ` -o "${outputFile}" "${url}"`;
      
      // Execute command with progress tracking
      const { stdout, stderr } = await execAsync(command);
      
      // Extract file size
      const stats = await fs.stat(outputFile);
      
      return {
        filePath: outputFile,
        fileSize: stats.size,
      };
    } catch (error) {
      console.error('Error downloading video:', error);
      throw new Error('Failed to download video');
    }
  }

  async cleanup(filePath: string) {
    try {
      await fs.unlink(filePath);
    } catch (error) {
      console.error('Error cleaning up file:', error);
    }
  }
}
```

export interface ConversionOptions {
  format?: string;
  quality?: number;
  width?: number;
  height?: number;
  compression?: number;
  fps?: number;
  bitrate?: string;
  gpuAcceleration?: boolean;
}

export interface ConversionResult {
  success: boolean;
  outputPath?: string;
  error?: string;
  originalSize?: number;
  newSize?: number;
  processingTime?: number;
}

// Enhanced image processing with GPU acceleration
export async function processImage(
  inputPath: string,
  outputPath: string,
  options: ConversionOptions
): Promise<ConversionResult> {
  const startTime = Date.now();
  
  try {
    const inputBuffer = await fs.readFile(inputPath);
    const originalSize = inputBuffer.length;

    let sharpInstance = sharp(inputBuffer, {
      // Enable GPU acceleration if available and requested
      sequentialRead: !options.gpuAcceleration,
    });

    // Resize if dimensions are provided
    if (options.width || options.height) {
      sharpInstance = sharpInstance.resize(
        options.width,
        options.height,
        { fit: 'inside', withoutEnlargement: true }
      );
    }

    // Set format
    if (options.format) {
      const formatOptions: any = {
        quality: options.quality || 80,
      };
      
      // Add compression level for formats that support it
      if (['png', 'webp', 'tiff'].includes(options.format)) {
        formatOptions.compressionLevel = options.compression || 6;
      }
      
      sharpInstance = sharpInstance.toFormat(options.format as any, formatOptions);
    } else {
      // Default to JPEG with quality
      sharpInstance = sharpInstance.jpeg({ quality: options.quality || 80 });
    }

    const outputBuffer = await sharpInstance.toBuffer();
    await fs.writeFile(outputPath, outputBuffer);

    return {
      success: true,
      outputPath,
      originalSize,
      newSize: outputBuffer.length,
      processingTime: Date.now() - startTime,
    };
  } catch (error) {
    console.error('Image processing error:', error);
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error',
      processingTime: Date.now() - startTime,
    };
  }
}

// Enhanced video processing with GPU acceleration
export async function processVideo(
  inputPath: string,
  outputPath: string,
  options: ConversionOptions
): Promise<ConversionResult> {
  const startTime = Date.now();
  
  return new Promise((resolve) => {
    try {
      // Get original file size
      fs.stat(inputPath)
        .then(stats => {
          const originalSize = stats.size;

          let command = ffmpeg(inputPath);

          // Enable GPU acceleration if requested
          if (options.gpuAcceleration) {
            // Use NVIDIA GPU acceleration if available
            command = command
              .videoCodec('h264_nvenc')
              .audioCodec('aac');
          } else {
            // Use CPU-based encoding
            command = command
              .videoCodec('libx264')
              .audioCodec('aac');
          }

          // Set output format
          if (options.format) {
            command = command.format(options.format);
          }

          // Set dimensions
          if (options.width || options.height) {
            command = command.size(`${options.width || '?'}x${options.height || '?'}`);
          }

          // Set quality
          if (options.quality) {
            command = command.videoBitrate(`${options.quality}k`);
          }

          // Set framerate
          if (options.fps) {
            command = command.fps(options.fps);
          }

          // Set bitrate
          if (options.bitrate) {
            command = command.bitrate(options.bitrate);
          }

          // Add performance optimizations
          command = command
            .preset('fast')
            .complexFilter([
              '[0:v]scale=trunc(iw/2)*2:trunc(ih/2)*2'
            ]);

          command
            .output(outputPath)
            .on('end', async () => {
              try {
                const outputStats = await fs.stat(outputPath);
                resolve({
                  success: true,
                  outputPath,
                  originalSize,
                  newSize: outputStats.size,
                  processingTime: Date.now() - startTime,
                });
              } catch (error) {
                resolve({
                  success: false,
                  error: 'Failed to get output file stats',
                  processingTime: Date.now() - startTime,
                });
              }
            })
            .on('error', (err) => {
              console.error('Video processing error:', err);
              resolve({
                success: false,
                error: err.message,
                processingTime: Date.now() - startTime,
              });
            })
            .run();
        })
        .catch(error => {
          console.error('Failed to get input file stats:', error);
          resolve({
            success: false,
            error: 'Failed to get input file stats',
            processingTime: Date.now() - startTime,
          });
        });
    } catch (error) {
      console.error('Video processing setup error:', error);
      resolve({
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error',
        processingTime: Date.now() - startTime,
      });
    }
  });
}

```typescript
// app/api/youtube/download/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth/next';
import { authOptions } from '@/lib/auth';
import { prisma } from '@/lib/db';
import { YouTubeDownloader } from '@/lib/youtube-downloader';
import { z } from 'zod';
import { redis } from '@/lib/redis';
import { v4 as uuidv4 } from 'uuid';

const DownloadRequestSchema = z.object({
  url: z.string().url(),
  format: z.enum(['video', 'audio']),
  resolution: z.string().optional(),
  audioQuality: z.number().optional(),
  outputFormat: z.string(),
});

export async function POST(request: NextRequest) {
  try {
    // Authenticate user
    const session = await getServerSession(authOptions);
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Parse and validate request
    const body = await request.json();
    const data = DownloadRequestSchema.parse(body);

    // Check rate limit
    const rateLimitKey = `youtube-download:${session.user.id}`;
    const currentCount = await redis.get(rateLimitKey);
    const rateLimit = 5; // 5 downloads per hour
    const window = 60 * 60; // 1 hour

    if (currentCount && parseInt(currentCount) >= rateLimit) {
      return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 });
    }

    // Increment rate limit counter
    await redis.incr(rateLimitKey);
    await redis.expire(rateLimitKey, window);

    // Create YouTube downloader instance
    const downloader = new YouTubeDownloader();

    // Get video info
    const videoInfo = await downloader.getVideoInfo(data.url);

    // Create download record
    const download = await prisma.youTubeDownload.create({
      data: {
        userId: session.user.id,
        videoId: videoInfo.id,
        title: videoInfo.title,
        duration: videoInfo.duration,
        thumbnail: videoInfo.thumbnail,
        format: data.format,
        resolution: data.resolution,
        audioQuality: data.audioQuality ? `${data.audioQuality}kbps` : undefined,
        outputFormat: data.outputFormat,
        status: 'pending',
      },
    });

    // Start download in background
    processDownload(download.id, data.url, data, downloader);

    return NextResponse.json({
      downloadId: download.id,
      status: 'pending',
      videoInfo: {
        title: videoInfo.title,
        duration: videoInfo.duration,
        thumbnail: videoInfo.thumbnail,
      },
    });
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json({ error: 'Invalid request', details: error.errors }, { status: 400 });
    }
    console.error('YouTube download error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

async function processDownload(
  downloadId: string,
  url: string,
  options: z.infer<typeof DownloadRequestSchema>,
  downloader: YouTubeDownloader
) {
  try {
    // Update status to processing
    await prisma.youTubeDownload.update({
      where: { id: downloadId },
      data: { status: 'processing' },
    });

    // Download video
    const result = await downloader.downloadVideo(url, options, async (progress) => {
      // Update progress in database
      await prisma.youTubeDownload.update({
        where: { id: downloadId },
        data: { progress },
      });
    });

    // Update status to completed
    await prisma.youTubeDownload.update({
      where: { id: downloadId },
      data: {
        status: 'completed',
        outputPath: result.filePath,
        fileSize: result.fileSize,
        progress: 100,
      },
    });

    // Schedule cleanup after 24 hours
    setTimeout(async () => {
      await downloader.cleanup(result.filePath);
      await prisma.youTubeDownload.update({
        where: { id: downloadId },
        data: { outputPath: null },
      });
    }, 24 * 60 * 60 * 1000); // 24 hours
  } catch (error) {
    // Update status to failed
    await prisma.youTubeDownload.update({
      where: { id: downloadId },
      data: {
        status: 'failed',
        error: error instanceof Error ? error.message : 'Unknown error',
      },
    });
  }
}

export async function GET(request: NextRequest) {
  try {
    // Authenticate user
    const session = await getServerSession(authOptions);
    if (!session?.user?.id) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // Get download ID from query params
    const { searchParams } = new URL(request.url);
    const downloadId = searchParams.get('id');

    if (!downloadId) {
      return NextResponse.json({ error: 'Download ID required' }, { status: 400 });
    }

    // Get download record
    const download = await prisma.youTubeDownload.findFirst({
      where: {
        id: downloadId,
        userId: session.user.id,
      },
    });

    if (!download) {
      return NextResponse.json({ error: 'Download not found' }, { status: 404 });
    }

    // If download is completed and file exists, return file
    if (download.status === 'completed' && download.outputPath) {
      try {
        const fileBuffer = await fs.readFile(download.outputPath);
        return new NextResponse(fileBuffer, {
          headers: {
            'Content-Type': 'application/octet-stream',
            'Content-Disposition': `attachment; filename="${download.title}.${download.outputFormat}"`,
          },
        });
      } catch (error) {
        // File might have been cleaned up
        return NextResponse.json({ error: 'File no longer available' }, { status: 404 });
      }
    }

    // Otherwise, return download status
    return NextResponse.json({
      id: download.id,
      status: download.status,
      progress: download.progress,
      error: download.error,
    });
  } catch (error) {
    console.error('YouTube download status error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

        body: JSON.stringify({
          url,
          format: downloadFormat,
          resolution: downloadFormat === 'video' ? selectedResolution : undefined,
          audioQuality: downloadFormat === 'audio' ? selectedAudioQuality : undefined,
          outputFormat,
        }),
      });

      if (!response.ok) {
        throw new Error('Failed to start download');
      }

      const data = await response.json();
      setDownloadStatus({
        id: data.downloadId,
        status: 'pending',
        progress: 0,
      });

      // Poll for download status
      pollDownloadStatus(data.downloadId);
    } catch (error) {
      toast({
        title: "Error",
        description: "Failed to start download. Please try again.",
        variant: "destructive",
      });
    } finally {
      setIsLoading(false);
    }
  }, [videoInfo, url, downloadFormat, selectedResolution, selectedAudioQuality, outputFormat, toast]);

  const pollDownloadStatus = useCallback((downloadId: string) => {
    const interval = setInterval(async () => {
      try {
        const response = await fetch(`/api/youtube/download?id=${downloadId}`);
        
        if (!response.ok) {
          clearInterval(interval);
          return;
        }

        const data = await response.json();
        
        setDownloadStatus({
          id: data.id,
          status: data.status,
          progress: data.progress,
          error: data.error,
          downloadUrl: data.status === 'completed' ? `/api/youtube/download?id=${downloadId}` : undefined,
        });

        if (data.status === 'completed' || data.status === 'failed') {
          clearInterval(interval);
          
          if (data.status === 'completed') {
            toast({
              title: "Download Complete",
              description: "Your YouTube video has been downloaded successfully.",
            });
          } else {
            toast({
              title: "Download Failed",
              description: data.error || "An unknown error occurred.",
              variant: "destructive",
            });
          }
        }
      } catch (error) {
        clearInterval(interval);
      }
    }, 2000);

    return () => clearInterval(interval);
  }, [toast]);

  const formatFileSize = (bytes: number) => {
    if (bytes === 0) return '0 Bytes';
    const k = 1024;
    const sizes = ['Bytes', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
  };

  const formatDuration = (seconds: number) => {
    const hours = Math.floor(seconds / 3600);
    const minutes = Math.floor((seconds % 3600) / 60);
    const secs = seconds % 60;
    return `${hours}:${minutes.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  };

  return (
    <div className="space-y-6">
      <Card>
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <Youtube className="h-5 w-5 text-red-500" />
            YouTube Downloader
          </CardTitle>
          <CardDescription>
            Download YouTube videos in up to 8K resolution and audio up to 320kbps quality
          </CardDescription>
        </CardHeader>
        <CardContent className="space-y-4">
          <div className="flex gap-2">
            <Input
              placeholder="https://www.youtube.com/watch?v=..."
              value={url}
              onChange={(e) => setUrl(e.target.value)}
              className="flex-1"
            />
            <Button onClick={fetchVideoInfo} disabled={isLoading || !url}>
              {isLoading ? <Loader2 className="h-4 w-4 animate-spin" /> : 'Get Info'}
            </Button>
          </div>

          <AnimatePresence>
            {videoInfo && (
              <motion.div
                initial={{ opacity: 0, height: 0 }}
                animate={{ opacity: 1, height: 'auto' }}
                exit={{ opacity: 0, height: 0 }}
                className="space-y-4"
              >
                <div className="flex gap-4 p-4 border rounded-lg">
                  {videoInfo.thumbnail && (
                    <img
                      src={videoInfo.thumbnail}
                      alt={videoInfo.title}
                      className="w-32 h-24 object-cover rounded"
                    />
                  )}
                  <div className="flex-1">
                    <h3 className="font-medium line-clamp-2">{videoInfo.title}</h3>
                    <p className="text-sm text-muted-foreground">{formatDuration(videoInfo.duration)}</p>
                  </div>
                </div>

                <Tabs value={downloadFormat} onValueChange={(value) => setDownloadFormat(value as 'video' | 'audio')}>
                  <TabsList className="grid w-full grid-cols-2">
                    <TabsTrigger value="video" className="flex items-center gap-2">
                      <Video className="h-4 w-4" />
                      Video
                    </TabsTrigger>
                    <TabsTrigger value="audio" className="flex items-center gap-2">
                      <Music className="h-4 w-4" />
                      Audio
                    </TabsTrigger>
                  </TabsList>

                  <TabsContent value="video" className="space-y-4">
                    <div>
                      <label className="text-sm font-medium mb-2 block">Resolution</label>
                      <Select value={selectedResolution} onValueChange={setSelectedResolution}>
                        <SelectTrigger>
                          <SelectValue />
                        </SelectTrigger>
                        <SelectContent>
                          {videoInfo.formats.video.map((format) => (
                            <SelectItem key={format.resolution} value={format.resolution}>
                              <div className="flex items-center justify-between w-full">
                                <span>{format.resolution}</span>
                                <Badge variant="outline">{format.codec}</Badge>
                              </div>
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>
                    </div>

                    <div>
                      <label className="text-sm font-medium mb-2 block">Output Format</label>
                      <Select value={outputFormat} onValueChange={setOutputFormat}>
                        <SelectTrigger>
                          <SelectValue />
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="mp4">MP4</SelectItem>
                          <SelectItem value="webm">WebM</SelectItem>
                          <SelectItem value="mkv">MKV</SelectItem>
                          <SelectItem value="avi">AVI</SelectItem>
                        </SelectContent>
                      </Select>
                    </div>
                  </TabsContent>

                  <TabsContent value="audio" className="space-y-4">
                    <div>
                      <label className="text-sm font-medium mb-2 block">Audio Quality</label>
                      <div className="space-y-2">
                        <div className="flex justify-between">
                          <span>Bitrate</span>
                          <span>{selectedAudioQuality}kbps</span>
                        </div>
                        <Slider
                          value={[selectedAudioQuality]}
                          onValueChange={(value) => setSelectedAudioQuality(value[0])}
                          max={320}
                          min={128}
                          step={32}
                          className="w-full"
                        />
                      </div>
                    </div>

                    <div>
                      <label className="text-sm font-medium mb-2 block">Output Format</label>
                      <Select value={outputFormat} onValueChange={setOutputFormat}>
                        <SelectTrigger>
                          <SelectValue />
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="mp3">MP3</SelectItem>
                          <SelectItem value="m4a">M4A</SelectItem>
                          <SelectItem value="wav">WAV</SelectItem>
                          <SelectItem value="flac">FLAC</SelectItem>
                        </SelectContent>
                      </Select>
                    </div>
                  </TabsContent>
                </Tabs>

                <Button
                  onClick={startDownload}
                  disabled={isLoading}
                  className="w-full"
                >
                  {isLoading ? (
                    <>
                      <Loader2 className="h-4 w-4 animate-spin mr-2" />
                      Processing...
                    </>
                  ) : (
                    <>
                      <Download className="h-4 w-4 mr-2" />
                      Download
                    </>
                  )}
                </Button>
              </motion.div>
            )}
          </AnimatePresence>

          <AnimatePresence>
            {downloadStatus && (
              <motion.div
                initial={{ opacity: 0, height: 0 }}
                animate={{ opacity: 1, height: 'auto' }}
                exit={{ opacity: 0, height: 0 }}
              >
                <Card>
                  <CardContent className="p-4">
                    <div className="flex items-center justify-between mb-2">
                      <div className="flex items-center gap-2">
                        {downloadStatus.status === 'pending' && <AlertCircle className="h-4 w-4" />}
                        {downloadStatus.status === 'processing' && <Loader2 className="h-4 w-4 animate-spin" />}
                        {downloadStatus.status === 'completed' && <CheckCircle className="h-4 w-4 text-green-500" />}
                        {downloadStatus.status === 'failed' && <XCircle className="h-4 w-4 text-red-500" />}
                        <span className="font-medium capitalize">{downloadStatus.status}</span>
                      </div>
                      <span className="text-sm text-muted-foreground">{downloadStatus.progress}%</span>
                    </div>
                    <Progress value={downloadStatus.progress} className="w-full" />
                    
                    {downloadStatus.status === 'completed' && downloadStatus.downloadUrl && (
                      <Button asChild className="w-full mt-2">
                        <a href={downloadStatus.downloadUrl} download>
                          Download File
                        </a>
                      </Button>
                    )}
                    
                    {downloadStatus.status === 'failed' && downloadStatus.error && (
                      <p className="text-sm text-red-500 mt-2">{downloadStatus.error}</p>
                    )}
                  </CardContent>
                </Card>
              </motion.div>
            )}
          </AnimatePresence>
        </CardContent>
      </Card>

      <div className="text-sm text-muted-foreground space-y-1">
        <p>• 8K videos may take longer to process depending on server resources</p>
        <p>• Files are automatically deleted after 24 hours for security</p>
        <p>• Download speeds may vary based on YouTube's availability</p>
      </div>
    </div>
  );
}
```

```dockerfile
# Dockerfile
FROM node:18-alpine AS base

# Install dependencies only when needed
FROM base AS deps
RUN apk add --no-cache libc6-compat
WORKDIR /app

# Install dependencies based on the preferred package manager
COPY package.json yarn.lock* package-lock.json* pnpm-lock.yaml* ./
RUN \
  if [ -f yarn.lock ]; then yarn --frozen-lockfile; \
  elif [ -f package-lock.json ]; then npm ci; \
  elif [ -f pnpm-lock.yaml ]; then yarn global add pnpm && pnpm i --frozen-lockfile; \
  else echo "Lockfile not found." && exit 1; \
  fi

# Install yt-dlp for YouTube downloads
RUN apk add --no-cache python3 py3-pip ffmpeg && \
  pip3 install yt-dlp

# Rebuild the source code only when needed
FROM base AS builder
WORKDIR /app
COPY --from=deps /app/node_modules ./node_modules
COPY . .

# Build the application
RUN npm run build

# Production image, copy all the files and run next
FROM base AS runner
WORKDIR /app

ENV NODE_ENV production

RUN addgroup --system --gid 1001 nodejs
RUN adduser --system --uid 1001 nextjs

COPY --from=builder /app/public ./public

# Set the correct permission for prerender cache
RUN mkdir .next
RUN chown nextjs:nodejs .next

# Automatically leverage output traces to reduce image size
COPY --from=builder --chown=nextjs:nodejs /app/.next/standalone ./
COPY --from=builder --chown=nextjs:nodejs /app/.next/static ./.next/static

USER nextjs

EXPOSE 3000

ENV PORT 3000
ENV HOSTNAME "0.0.0.0"

CMD ["node", "server.js"]
```
