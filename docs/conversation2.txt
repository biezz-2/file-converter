### 2.3 File Processing Worker (`workers/conversion.worker.ts`)

```typescript
import { Worker } from 'bullmq'
import { redisConnection, conversionQueue } from '../lib/queue'
import { processImage, processVideo, processArchive } from '../lib/file-processing'
import { prisma } from '../lib/db'
import { uploadToS3, deleteFromS3 } from '../lib/storage'
import { createClamAV } from 'clamscan'
import pino from 'pino'

const logger = pino({ name: 'conversion-worker' })

const clamav = await createClamAV({
  clamdscan: {
    socket: process.env.CLAMAV_SOCKET || '/var/run/clamav/clamd.sock',
    timeout: 30000,
  },
})

export const conversionWorker = new Worker(
  'conversion',
  async (job) => {
    const startTime = Date.now()
    const { fileId, userId, originalPath, outputPath, options, originalName, apiKeyId } = job.data as ConversionJob

    try {
      // 1. Validate user quota
      const quota = await prisma.userQuota.findUnique({ where: { userId } })
      if (quota && quota.conversionsToday >= quota.maxDailyConversions) {
        throw new Error('Daily conversion limit exceeded')
      }

      // 2. Scan for viruses
      const scanResult = await clamav.scanFile(originalPath)
      if (scanResult.isInfected) {
        throw new Error(`File is infected: ${scanResult.viruses.join(', ')}`)
      }

      // 3. Update progress
      await job.updateProgress(20)

      // 4. Process based on file type
      const fileType = getFileType(originalName)
      let result

      switch (fileType) {
        case 'image':
          result = await processImage(originalPath, outputPath, options)
          break
        case 'video':
          result = await processVideo(originalPath, outputPath, options)
          break
        case 'archive':
          result = await processArchive(originalPath, outputPath, options)
          break
        default:
          throw new Error('Unsupported file type')
      }

      if (!result.success) {
        throw new Error(result.error)
      }

      await job.updateProgress(60)

      // 5. Upload to S3
      const s3Key = `conversions/${userId}/${fileId}/${path.basename(outputPath)}`
      const s3Url = await uploadToS3(outputPath, s3Key, result.newType || options.outputFormat)

      await job.updateProgress(80)

      // 6. Update database
      const conversion = await prisma.conversion.create({
        data: {
          userId,
          apiKeyId,
          originalName,
          originalType: fileType,
          originalSize: result.originalSize!,
          newName: path.basename(outputPath),
          newType: options.outputFormat,
          newSize: result.newSize,
          status: 'completed',
          processingTime: Date.now() - startTime,
          options: JSON.stringify(options),
        },
      })

      // 7. Update quota
      await prisma.userQuota.update({
        where: { userId },
        data: {
          conversionsToday: { increment: 1 },
        },
      })

      await job.updateProgress(100)

      logger.info({ 
        conversionId: conversion.id, 
        userId, 
        fileName: originalName,
        processingTime: Date.now() - startTime 
      }, 'Conversion completed')

      return { conversionId: conversion.id, s3Url, size: result.newSize }

    } catch (error) {
      logger.error({ 
        jobId: job.id, 
        userId, 
        error: error instanceof Error ? error.message : 'Unknown error',
        stack: error instanceof Error ? error.stack : undefined
      }, 'Conversion failed')

      await prisma.conversion.create({
        data: {
          userId,
          apiKeyId,
          originalName,
          originalType: getFileType(originalName),
          originalSize: job.data.originalSize,
          status: 'failed',
          error: error instanceof Error ? error.message : 'Unknown error',
        },
      })

      throw error
    } finally {
      // Cleanup temp files
      try {
        await fs.unlink(originalPath)
        await fs.unlink(outputPath)
      } catch (cleanupError) {
        logger.warn({ cleanupError }, 'Failed to cleanup temp files')
      }
    }
  },
  {
    connection: redisConnection,
    concurrency: parseInt(process.env.CONVERSION_WORKER_CONCURRENCY || '2'),
    limiter: {
      max: 10,
      duration: 1000,
    },
  }
)

// Error handling
conversionWorker.on('failed', (job, err) => {
  logger.error({ jobId: job?.id, err }, 'Worker job failed')
})

conversionWorker.on('error', (err) => {
  logger.error({ err }, 'Worker error')
})
```

### 3.3 YouTube Worker (`services/youtube-downloader/worker.py`)

```python
import os
import redis
import json
import yt_dlp
import shutil
from pathlib import Path
from typing import Dict, Any
import logging
from dataclasses import dataclass
import psutil

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

@dataclass
class YouTubeJob:
    job_id: str
    user_id: str
    video_url: str
    format: str
    resolution: str
    audio_quality: str
    output_format: str

class YouTubeDownloader:
    def __init__(self, redis_url: str):
        self.redis = redis.Redis.from_url(redis_url, decode_responses=True)
        self.download_path = Path("/app/downloads")
        self.download_path.mkdir(exist_ok=True)
        
    def get_video_info(self, url: str) -> Dict[str, Any]:
        """Get video metadata without downloading"""
        ydl_opts = {
            'quiet': True,
            'no_warnings': True,
            'extract_flat': True,
        }
        
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            info = ydl.extract_info(url, download=False)
            return ydl.sanitize_info(info)
    
    def download_video(self, job: YouTubeJob) -> Dict[str, Any]:
        """Download video with specified quality"""
        output_filename = f"{job.job_id}_%(title)s.%(ext)s"
        output_path = self.download_path / output_filename
        
        ydl_opts = {
            'outtmpl': str(output_path),
            'progress_hooks': [self._progress_hook(job.job_id)],
            'format': self._build_format_selector(job),
            'postprocessors': self._build_postprocessors(job),
            'merge_output_format': job.output_format,
            # Quality settings
            'prefer_ffmpeg': True,
            'ffmpeg_location': '/usr/bin/ffmpeg',
            # Performance
            'retries': 3,
            'fragment_retries': 3,
            'skip_unavailable_fragments': True,
            # Limit concurrent downloads
            'max_downloads': 1,
            'throttledrate': '1M', # Throttle to 1MB/s per download
        }
        
        # Add cookies for age-restricted content if available
        cookies_file = Path("/app/config/cookies.txt")
        if cookies_file.exists():
            ydl_opts['cookiefile'] = str(cookies_file)
        
        try:
            with yt_dlp.YoutubeDL(ydl_opts) as ydl:
                info = ydl.download([job.video_url])
                return {
                    'status': 'completed',
                    'file_path': str(output_path),
                    'file_size': output_path.stat().st_size,
                    'title': info.get('title'),
                    'duration': info.get('duration'),
                }
        except Exception as e:
            logger.error(f"Download failed: {str(e)}")
            raise
    
    def _build_format_selector(self, job: YouTubeJob) -> str:
        """Build yt-dlp format selector"""
        if job.format == 'audio':
            # Audio only
            if job.audio_quality == '320kbps':
                return 'bestaudio[ext=m4a]/bestaudio[ext=webm]/bestaudio'
            else:
                return f'bestaudio[abr<={job.audio_quality.replace("kbps", "")}]'
        else:
            # Video with specific resolution
            if job.resolution == '4320p': # 8K
                return 'bestvideo[height<=4320][vcodec^=av01]/bestvideo[height<=4320][vcodec^=vp9]/bestvideo'
            elif job.resolution == '2160p': # 4K
                return 'bestvideo[height<=2160][vcodec^=av01]/bestvideo[height<=2160][vcodec^=vp9]/bestvideo'
            elif job.resolution == '1080p':
                return 'bestvideo[height<=1080]/bestvideo'
            else:
                return f'bestvideo[height<={job.resolution.replace("p", "")}]'
    
    def _build_postprocessors(self, job: YouTubeJob) -> list:
        """Build postprocessors for format conversion"""
        postprocessors = []
        
        if job.format == 'audio':
            if job.output_format == 'mp3':
                postprocessors.append({
                    'key': 'FFmpegExtractAudio',
                    'preferredcodec': 'mp3',
                    'preferredquality': job.audio_quality.replace('kbps', ''),
                })
            elif job.output_format == 'm4a':
                postprocessors.append({
                    'key': 'FFmpegExtractAudio',
                    'preferredcodec': 'm4a',
                    'preferredquality': 'best',
                })
        
        return postprocessors
    
    def _progress_hook(self, job_id: str):
        """Hook to report progress to Redis"""
        def hook(d):
            if d['status'] == 'downloading':
                progress = {
                    'downloaded_bytes': d.get('downloaded_bytes', 0),
                    'total_bytes': d.get('total_bytes') or d.get('total_bytes_estimate', 0),
                    'speed': d.get('speed', 0),
                    'eta': d.get('eta', 0),
                    'percent': d.get('_percent_str', '0%'),
                }
                # Calculate percentage
                if progress['total_bytes'] > 0:
                    pct = int((progress['downloaded_bytes'] / progress['total_bytes']) * 100)
                    self.redis.hset(f"job:{job_id}:progress", mapping={'progress': pct})
                    self.redis.publish(f"job:{job_id}:updates", json.dumps({'progress': pct}))
        
        return hook
    
    def cleanup_old_files(self):
        """Cleanup files older than 24 hours"""
        now = time.time()
        for file in self.download_path.glob("*"):
            if file.is_file() and now - file.stat().st_mtime > 86400:
                file.unlink()
                logger.info(f"Cleaned up old file: {file.name}")

def main():
    redis_url = os.getenv('REDIS_URL', 'redis://redis:6379')
    downloader = YouTubeDownloader(redis_url)
    
    # Cleanup old files every hour
    def periodic_cleanup():
        while True:
            time.sleep(3600)
            downloader.cleanup_old_files()
    
    import threading
    cleanup_thread = threading.Thread(target=periodic_cleanup, daemon=True)
    cleanup_thread.start()
    
    # Process jobs
    while True:
        try:
            # Get job from queue
            job_data = downloader.redis.blpop('youtube:jobs', timeout=1)
            if job_data:
                _, data = job_data
                job_dict = json.loads(data)
                job = YouTubeJob(**job_dict)
                
                # Update status to processing
                downloader.redis.hset(f"job:{job.job_id}:status", mapping={'status': 'processing'})
                
                try:
                    result = downloader.download_video(job)
                    result['status'] = 'completed'
                    downloader.redis.hset(f"job:{job.job_id}:result", mapping=result)
                    downloader.redis.publish(f"job:{job.job_id}:complete", json.dumps(result))
                except Exception as e:
                    logger.error(f"Job {job.job_id} failed: {str(e)}")
                    downloader.redis.hset(f"job:{job.job_id}:error", mapping={
                        'status': 'failed',
                        'error': str(e)
                    })
                
        except Exception as e:
            logger.error(f"Worker error: {str(e)}")
            time.sleep(5)

if __name__ == '__main__':
    main()
```

import { NextRequest, NextResponse } from 'next/server'

// Initialize Rate Limiter

// API Key Validation
export async function validateApiKey(request: NextRequest) {
  const authHeader = request.headers.get('authorization')
  if (!authHeader?.startsWith('Bearer ')) {
    return null
  }

  const apiKey = authHeader.replace('Bearer ', '')
  const keyHash = createHash('sha256').update(apiKey).digest('hex')

  const apiKeyRecord = await prisma.apiKey.findFirst({
    where: {
      keyHash,
      isActive: true,
      expiresAt: { gte: new Date() },
    },
    include: { user: true },
  })

  if (!apiKeyRecord) {
    return null
  }

  // Increment usage
  await prisma.apiKey.update({
    where: { id: apiKeyRecord.id },
    data: { usageCount: { increment: 1 }, lastUsed: new Date() },
  })

  return apiKeyRecord
}

// File Validation Schema
export const FileUploadSchema = z.object({
  file: z.any().refine(
    (file) => {
      // Validate file magic numbers
      const buffer = file.slice(0, 4)
      const magicNumbers = {
        'image/jpeg': ['ffd8ffe0', 'ffd8ffe1', 'ffd8ffe2'],
        'image/png': ['89504e47'],
        'video/mp4': ['00000018', '66747970'],
        // Add more signatures
      }
      
      const fileMagic = Array.from(new Uint8Array(buffer))
        .map((b) => b.toString(16).padStart(2, '0'))
        .join('')
      
      return Object.values(magicNumbers).some((signatures) =>
        signatures.some((sig) => fileMagic.startsWith(sig))
      )
    },
    'Invalid file signature'
  ),
})

// EnterpriseAuth Middleware
export const withEnterpriseAuth = (handler: Function, options: {
  requireApiKey?: boolean
  requireAdmin?: boolean
  scopes?: string[]
} = {}) => {
  return async (request: NextRequest, ...args: any[]) => {
    try {
      // Check API Key first
      const apiKey = await validateApiKey(request)
      if (apiKey) {
        if (options.scopes && !options.scopes.some(scope => apiKey.permissions.includes(scope))) {
          return NextResponse.json({ error: 'Insufficient permissions' }, { status: 403 })
        }
        
        // Rate limit by API key
        const { success } = await ratelimit.limit(`api:${apiKey.id}`)
        if (!success) {
          return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 })
        }

        return handler(request, { user: apiKey.user, apiKey }, ...args)
      }

      // Fall back to session auth
      const token = await getToken({ req: request })
      if (!token?.sub) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
      }

      const user = await prisma.user.findUnique({
        where: { id: token.sub },
        include: { quota: true },
      })

      if (!user || !user.isActive) {
        return NextResponse.json({ error: 'User not active' }, { status: 401 })
      }

      if (options.requireAdmin && !user.isAdmin) {
        return NextResponse.json({ error: 'Admin required' }, { status: 403 })
      }

      // Rate limit by user
      const { success } = await ratelimit.limit(`user:${user.id}`)
      if (!success) {
        return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 })
      }

      return handler(request, { user }, ...args)
    } catch (error) {
      console.error('Auth middleware error:', error)
      return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
    }
  }
}

import { NextRequest, NextResponse } from 'next/server'
import { withEnterpriseAuth } from '@/middleware/security'
import { put } from '@vercel/blob' // or AWS S3

const ConvertRequestSchema = z.object({
  outputFormat: z.string().min(1),
  quality: z.number().min(1).max(100).optional(),
  compression: z.number().min(0).max(9).optional(),
  resize: z.boolean().optional(),
  width: z.number().min(100).max(8192).optional(),
  height: z.number().min(100).max(8192).optional(),
  gpuAcceleration: z.boolean().optional(),
})

export const POST = withEnterpriseAuth(
  async (request: NextRequest, { user }) => {
    try {
      // Check quota
      const quota = await prisma.userQuota.findUnique({ where: { userId: user.id } })
      if (quota && quota.conversionsToday >= quota.maxDailyConversions) {
        return NextResponse.json({ error: 'Daily conversion limit exceeded' }, { status: 429 })
      }

      const formData = await request.formData()
      const file = formData.get('file') as File
      const options = ConvertRequestSchema.parse(JSON.parse(formData.get('options') as string))

      if (!file) {
        return NextResponse.json({ error: 'No file provided' }, { status: 400 })
      }

      // Check file size
      if (file.size > quota?.maxFileSize || file.size > 2 * 1024 * 1024 * 1024) {
        return NextResponse.json({ error: 'File too large' }, { status: 413 })
      }

      // Save file to temp storage (in production, use direct S3 upload)
      const fileId = uuidv4()
      const tempDir = join(process.cwd(), 'temp', fileId)
      await mkdir(tempDir, { recursive: true })
      
      const originalPath = join(tempDir, file.name)
      const outputPath = join(tempDir, `output.${options.outputFormat}`)
      
      await writeFile(originalPath, Buffer.from(await file.arrayBuffer()))

      // Add to queue
      const job = await conversionQueue.add('conversion', {
        fileId,
        userId: user.id,
        originalPath,
        outputPath,
        options,
        originalName: file.name,
        originalSize: file.size,
      })

      // Return job ID for polling
      return NextResponse.json({
        jobId: job.id,
        status: 'queued',
        message: 'File queued for processing',
      })

    } catch (error) {
      if (error instanceof z.ZodError) {
        return NextResponse.json({ error: 'Invalid options', details: error.errors }, { status: 400 })
      }
      console.error('Convert error:', error)
      return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
    }
  },
  { requireApiKey: false } // Optional: require API key
)

// GET endpoint for job status
export const GET = withEnterpriseAuth(
  async (request: NextRequest, { user }) => {
    const { searchParams } = new URL(request.url)
    const jobId = searchParams.get('jobId')

    if (!jobId) {
      return NextResponse.json({ error: 'jobId required' }, { status: 400 })
    }

    const job = await conversionQueue.getJob(jobId)
    if (!job) {
      return NextResponse.json({ error: 'Job not found' }, { status: 404 })
    }

    // Check if user owns this job
    if (job.data.userId !== user.id) {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
    }

    return NextResponse.json({
      jobId: job.id,
      status: job.status,
      progress: job.progress,
      result: job.returnvalue,
      failedReason: job.failedReason,
    })
  }

import { withEnterpriseAuth } from '@/middleware/security'

export const POST = withEnterpriseAuth(async (request: NextRequest) => {
  const body = await request.json()
  const { url } = body

  if (!ytdl.validateURL(url)) {
    return NextResponse.json({ error: 'Invalid YouTube URL' }, { status: 400 })
  }

  try {
    const info = await ytdl.getInfo(url)
    
    // Parse available formats
    const videoFormats = info.formats
      .filter(f => f.hasVideo && f.height)
      .sort((a, b) => (b.height || 0) - (a.height || 0))
      .map(f => ({
        quality: f.qualityLabel,
        resolution: `${f.height}p`,
        fps: f.fps,
        codec: f.codecs,
        filesize: f.contentLength,
      }))
      .filter((f, i, arr) => arr.findIndex(x => x.resolution === f.resolution) === i) // dedupe
    
    const audioFormats = info.formats
      .filter(f => f.hasAudio && !f.hasVideo && f.audioBitrate)
      .sort((a, b) => (b.audioBitrate || 0) - (a.audioBitrate || 0))
      .map(f => ({
        quality: f.audioBitrate,
        format: f.audioCodec,
        filesize: f.contentLength,
      }))

    return NextResponse.json({
      title: info.videoDetails.title,
      author: info.videoDetails.author.name,
      lengthSeconds: info.videoDetails.lengthSeconds,
      thumbnail: info.videoDetails.thumbnails[0]?.url,
      videoFormats,
      audioFormats,
    })
  } catch (error) {
    console.error('YouTube info error:', error)
    return NextResponse.json({ error: 'Failed to fetch video info' }, { status: 500 })
  }

import { NextRequest, NextResponse } from 'next/server'
import { withEnterpriseAuth } from '@/middleware/security'
import { youtubeQueue } from '@/lib/queue'
import { prisma } from '@/lib/db'
import { z } from 'zod'

const DownloadRequestSchema = z.object({
  url: z.string().url(),
  format: z.enum(['video', 'audio']),
  resolution: z.string().optional(),
  audioQuality: z.string().optional(),
  outputFormat: z.string(),
  webhookUrl: z.string().url().optional(),
})

export const POST = withEnterpriseAuth(async (request: NextRequest, { user }) => {
  try {
    // Check download quota
    const quota = await prisma.userQuota.findUnique({ where: { userId: user.id } })
    if (quota && quota.downloadsToday >= quota.maxDailyDownloads) {
      return NextResponse.json({ error: 'Daily download limit exceeded' }, { status: 429 })
    }

    const body = await request.json()
    const data = DownloadRequestSchema.parse(body)

    // Validate resolution for format
    if (data.format === 'video' && !data.resolution) {
      return NextResponse.json({ error: 'Resolution required for video format' }, { status: 400 })
    }
    
    if (data.format === 'audio' && !data.audioQuality) {
      return NextResponse.json({ error: 'Audio quality required for audio format' }, { status: 400 })
    }

    // Add to queue
    const job = await youtubeQueue.add('download', {
      userId: user.id,
      videoUrl: data.url,
      format: data.format,
      resolution: data.resolution,
      audioQuality: data.audioQuality,
      outputFormat: data.outputFormat,
      webhookUrl: data.webhookUrl,
    })

    // Create download record
    await prisma.youTubeDownload.create({
      data: {
        id: job.id!,
        userId: user.id,
        videoUrl: data.url,
        format: data.format,
        resolution: data.resolution,
        audioQuality: data.audioQuality,
        outputFormat: data.outputFormat,
        status: 'queued',
      },
    })

    return NextResponse.json({
      jobId: job.id,
      status: 'queued',
      message: 'Download queued successfully',
      estimatedTime: '2-5 minutes for 8K content',
    })

  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json({ error: 'Invalid request', details: error.errors }, { status: 400 })
    }
    console.error('Download error:', error)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
})

   # Ensure NVIDIA Docker runtime

## 10. API Documentation (OpenAPI)

Buat file `openapi.yml` di root project:

```yaml
openapi: 3.0.3
info:
  title: File Converter Enterprise API
  version: 1.0.0
  description: Enterprise-grade file conversion and YouTube download API

servers:
  - url: https://api.your-domain.com
    description: Production

security:
  - bearerAuth: []
  - apiKey: []

paths:
  /api/convert:
    post:
      summary: Queue file conversion
      requestBody:
        content:
          multipart/form-data:
            schema:
              type: object
              properties:
                file:
                  type: string
                  format: binary
                options:
                  type: string
      responses:
        '202':
          description: Accepted
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/JobResponse'

  /api/youtube/download:
    post:
      summary: Queue YouTube download
      requestBody:
        content:
          application/json:
            schema:
              $ref: '#/components/schemas/YouTubeDownloadRequest'
      responses:
        '202':
          description: Accepted
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/JobResponse'

components:
  securitySchemes:
    bearerAuth:
      type: http
      scheme: bearer
    apiKey:
      type: apiKey
      in: header
      name: X-API-Key

  schemas:
    JobResponse:
      type: object
      properties:
        jobId:
          type: string
        status:
          type: string
          enum: [queued, processing, completed, failed]
        progress:
          type: number
```
